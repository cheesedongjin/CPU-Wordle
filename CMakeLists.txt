cmake_minimum_required(VERSION 3.16)
project(cpu_wordle C)
set(CMAKE_C_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSON REQUIRED libcjson)
pkg_check_modules(UTF8 REQUIRED libutf8proc)
find_package(PostgreSQL REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_library(engine src/engine.c)
target_include_directories(engine PUBLIC src ${UTF8_INCLUDE_DIRS} ${JSON_INCLUDE_DIRS})
target_link_libraries(engine PUBLIC ${UTF8_LIBRARIES} ${JSON_LIBRARIES} OpenSSL::Crypto)
add_executable(wordle src/server.c src/storage.c)
target_include_directories(wordle PRIVATE ${PostgreSQL_INCLUDE_DIRS})
target_link_libraries(wordle PRIVATE engine civetweb ${PostgreSQL_LIBRARIES} Threads::Threads)
add_executable(test_engine tests/test_engine.c)
target_link_libraries(test_engine PRIVATE engine)
enable_testing()
add_test(NAME engine COMMAND test_engine WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
